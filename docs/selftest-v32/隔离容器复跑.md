# 本地隔离容器复跑

这是[基线复跑](自测报告.md#可复核的基线复跑)使用的设置。它要求工作区的 `docker-local`（含 `Dockerfile`、`compose.yml`、`regression.yml`）、Docker Compose、`jq`，以及包含 Chromium 和 Playwright 的本地 `i3-composer-e2e:local` 镜像。Docker 镜像名可以用 `I3_E2E_IMAGE` 替换，但浏览器路径和模块布局必须一致。

在仓库根目录用 Bash 执行。先导出 `I3_COMPOSE_DIR`（工作区 `docker-local` 的绝对路径）、`I3_HTTP_PORT`（空闲端口）和 `I3_TEST_SUBNET`（未用子网）。默认归档当前提交；要复核报告中的基线，设置 `ACCEPTANCE_REF=5a88e9fcecb6f8273dbb290ac62b4d7d855ad145`。构建和运行都使用归档，不读主 checkout 的代码。复跑创建自己的项目和账本；退出后停止该项目容器，保留本轮数据卷和证据。

```bash
(
  set -euo pipefail
  : "${I3_COMPOSE_DIR:?请给出工作区 docker-local 的绝对路径}"
  : "${I3_HTTP_PORT:?请选一个空闲的宿主机端口}"
  : "${I3_TEST_SUBNET:?请选一段未使用的 Docker 子网}"

  worktree_status=$(git status --porcelain --untracked-files=all)
  if [ -n "$worktree_status" ]; then
    printf '%s\n' '工作树含未提交改动，结果不代表任何提交。' "$worktree_status" >&2
    exit 1
  fi
  run_root=$(mktemp -d "${TMPDIR:-/tmp}/i3-replay.XXXXXX")
  printf '本轮证据目录：%s\n' "$run_root"
  commit=$(git rev-parse "${ACCEPTANCE_REF:-HEAD}^{commit}")
  printf '%s\n' "$commit" > "$run_root/commit.txt"
  printf '%s' "$worktree_status" > "$run_root/git-status.txt"
  mkdir "$run_root/source"
  I3_SOURCE="$run_root/source"
  git archive "$commit" | tar -x -C "$I3_SOURCE"
  project="i3-replay-$(date +%s)-$$"
  I3_APP_IMAGE="i3-composer:$project"
  I3_LEDGER_PREFIX="$project/"
  e2e_image="${I3_E2E_IMAGE:-i3-composer-e2e:local}"
  export I3_SOURCE I3_APP_IMAGE I3_LEDGER_PREFIX I3_HTTP_PORT I3_TEST_SUBNET
  docker build -f "$I3_COMPOSE_DIR/Dockerfile" --target app -t "$I3_APP_IMAGE" "$I3_SOURCE"

  cat > "$run_root/acceptance.yml" <<'YAML'
services:
  app:
    working_dir: /tmp
    environment:
      ADDR: 127.0.0.1:8080
      WEB_DIR: /app/web/dist
      LEDGER_PREFIX: ${I3_LEDGER_PREFIX}
      REQUIRE_PUBSUB: '1'
  app-channel:
    image: ${I3_E2E_IMAGE:-i3-composer-e2e:local}
    network_mode: service:app
    entrypoint: [node, -e]
    command:
      - |
        const http = require('node:http');
        http.createServer((req, res) => {
          const proxy = http.request({
            host: '127.0.0.1', port: 8080, path: req.url, method: req.method, headers: req.headers
          }, upstream => {
            res.writeHead(upstream.statusCode, upstream.headers);
            upstream.pipe(res);
          });
          proxy.on('error', error => {
            const pending = error.code === 'ECONNREFUSED' && req.url === '/dapr/config';
            if (!res.headersSent) res.writeHead(pending ? 404 : 503);
            res.end();
          });
          req.on('aborted', () => proxy.destroy());
          req.pipe(proxy);
        }).listen(8080, '127.0.0.2');
    healthcheck:
      test: ["CMD", "node", "-e", "const socket=require('node:net').connect(8080,'127.0.0.2'); socket.once('connect',()=>socket.end()); socket.once('error',()=>process.exit(1))"]
      interval: 1s
      timeout: 2s
      retries: 20
  daprd:
    depends_on:
      app-channel:
        condition: service_healthy
    command:
      - /daprd
      - --app-id=i3-composer
      - --app-port=8080
      - --app-channel-address=127.0.0.2
      - --dapr-http-port=3500
      - --dapr-grpc-port=50001
      - --placement-host-address=placement:50005
      - --scheduler-host-address=scheduler:50006
      - --resources-path=/components
configs:
  subscription:
    file: ${I3_SOURCE}/components/subscription.yaml
YAML

  compose() {
    docker compose -p "$project" -f "$I3_COMPOSE_DIR/compose.yml" \
      -f "$I3_COMPOSE_DIR/regression.yml" -f "$run_root/acceptance.yml" "$@"
  }
  trap 'compose down --remove-orphans' EXIT
  compose up -d --no-build --wait --wait-timeout 120
  compose exec -T app wget -qO- http://127.0.0.1:8080/api/health |
    tee "$run_root/health.json" | jq -e '.pubsub.checked and .pubsub.ok'

  app_container=$(compose ps -q app)
  docker run --rm --init --shm-size 512m --network "container:$app_container" \
    -e BASE=http://127.0.0.1:8080 -e RUN_TAG="$(date +%s)" \
    -e NODE_PATH=/tmp/regression/node_modules \
    -v "$I3_SOURCE:/source:ro" -v "$run_root:/evidence" --entrypoint sh "$e2e_image" -ec '
      cd /source/web/e2e
      chrome="/Users/nukzhang/Library/Caches/ms-playwright/chromium-1217/chrome-mac-arm64/Google Chrome for Testing.app/Contents/MacOS/Google Chrome for Testing"
      mkdir -p "$(dirname "$chrome")"
      ln -s "$CHROMIUM_PATH" "$chrome"
      for suite in acceptance_v32 acceptance_exec acceptance_flow acceptance_full acceptance_v26; do
        export SHOTS="/evidence/$suite-shots"
        if node "$suite.cjs" > "/evidence/$suite.log" 2>&1; then
          grep "^SUMMARY" "/evidence/$suite.log"
        else
          cat "/evidence/$suite.log" >&2
          exit 1
        fi
      done
    '
)
```

临时转发仅解决本地 Dapr 回调先于应用就绪的问题；应用就绪后的 Pub/Sub、工作流、状态存储和浏览器均是真实执行。报告使用各套实际输出的 `SUMMARY`，不能把一个提交的结果当成另一个提交的状态。
